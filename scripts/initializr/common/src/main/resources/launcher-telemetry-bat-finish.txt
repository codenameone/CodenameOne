:finish
set "CN1_EXIT_CODE=%errorlevel%"
REM Why it stopped. build.sh reads Maven's output for this; cmd cannot tee it
REM without hiding it from you, so here the class comes from what was checked
REM before the build: no JDK, or one older than this project's Java level, are
REM the two failures a first build hits most and both are known without the
REM log. Anything else is reported as build_failed. cmd cannot trap Ctrl+C
REM either, so an interrupted build sends no exit report.
set "CN1_REASON=build_failed"
if "%CN1_JAVA%"=="none" set "CN1_REASON=no_jdk"
set "CN1_JMAJ=0"
set /a CN1_JMAJ=%CN1_JAVA% >nul 2>&1
if not "%CN1_JAVA%"=="none" if %CN1_JMAJ% GTR 0 if %CN1_JMAJ% LSS %CN1_MIN_JAVA% set "CN1_REASON=java_too_old"
if "%CN1_EXIT_CODE%"=="0" set "CN1_REASON=ok"
set "CN1_SECS="
if defined CN1_START call :cn1_now
if defined CN1_START if defined CN1_NOW set /a "CN1_SECS=CN1_NOW-CN1_START"
if defined CN1_SECS if %CN1_SECS% LSS 0 set /a "CN1_SECS+=86400"
if /i not "%CN1_TARGET%"=="help" call :cn1_event "step=exit" "exit=%CN1_EXIT_CODE%" "reason=%CN1_REASON%" "secs=%CN1_SECS%"
popd
exit /b %CN1_EXIT_CODE%

REM Seconds since midnight from %TIME% (H:MM:SS.cc, or with a comma before the
REM hundredths in some locales). "1%%a-100" strips the leading zero that set /a
REM would otherwise read as octal; a leading space in the hour becomes a zero.
:cn1_now
set "CN1_NOW="
for /f "tokens=1-3 delims=:.," %%a in ("%TIME: =0%") do set /a "CN1_NOW=(1%%a-100)*3600+(1%%b-100)*60+(1%%c-100)" 2>nul
goto :eof

REM Fire-and-forget: three seconds at most, and a failure is never an error.
:cn1_event
if "%CN1_TELEMETRY%"=="0" goto :eof
where curl.exe >nul 2>&1 || goto :eof
curl.exe -s -o nul -m 3 -X POST "%CN1_EVENTS_URL%" --data-urlencode "pkg=%CN1_PROJECT_ID%" --data-urlencode "target=%CN1_TARGET%" --data-urlencode "os=windows" --data-urlencode "java=%CN1_JAVA%" --data-urlencode %1 --data-urlencode %2 --data-urlencode %3 --data-urlencode %4 >nul 2>&1
goto :eof
