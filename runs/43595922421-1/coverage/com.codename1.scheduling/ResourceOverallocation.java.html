package com.codename1.scheduling;

/// A stretch of time in which the tasks assigned to a resource need more of
/// it than there is.
public final class ResourceOverallocation {
    private final String resourceId;
    private final long start;
    private final long end;
    private final float peak;
    private final float capacity;

    ResourceOverallocation(String resourceId, long start, long end, float peak, float capacity) {
        this.resourceId = resourceId;
        this.start = start;
        this.end = end;
        this.peak = peak;
        this.capacity = capacity;
    }

    public String getResourceId() {
        return resourceId;
    }

    public long getStart() {
        return start;
    }

    public long getEnd() {
        return end;
    }

    /// The most units asked for at once in the stretch.
    public float getPeak() {
        return peak;
    }

    /// The units the resource has.
    public float getCapacity() {
        return capacity;
    }
}
