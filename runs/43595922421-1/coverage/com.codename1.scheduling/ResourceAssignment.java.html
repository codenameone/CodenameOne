package com.codename1.scheduling;

/// How much of a resource a task uses while it runs.
public final class ResourceAssignment {
    private final String resourceId;
    private final float units;

    /// `units` of 1 is the whole of one resource for the length of the task.
    public ResourceAssignment(String resourceId, float units) {
        if (resourceId == null) {
            throw new IllegalArgumentException("resourceId");
        }
        if (units <= 0) {
            throw new IllegalArgumentException("units");
        }
        this.resourceId = resourceId;
        this.units = units;
    }

    public String getResourceId() {
        return resourceId;
    }

    public float getUnits() {
        return units;
    }
}
