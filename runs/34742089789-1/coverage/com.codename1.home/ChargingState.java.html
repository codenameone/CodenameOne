package com.codename1.home;

/// Whether an accessory's battery is charging, for [Trait#BATTERY_CHARGING].
/// Read-only.
public enum ChargingState {

    /// The accessory could not say.
    ///
    /// **HomeKit never produces this** -- its characteristic has no such
    /// value.
    UNKNOWN,

    /// On battery.
    NOT_CHARGING,

    /// Charging.
    CHARGING,

    /// On external power with the battery already full.
    ///
    /// **HomeKit never produces this**; it reports [#NOT_CHARGING] for a
    /// full battery sitting on a charger, which is indistinguishable from one
    /// that is running down.
    FULL,

    /// The battery is not rechargeable -- a coin cell in a sensor.
    ///
    /// **Matter and Google Home never produce this.** Matter expresses it by
    /// the accessory simply not implementing the rechargeable feature, which
    /// this API cannot see from the charge attribute alone.
    NOT_CHARGEABLE;

    /// Reads a charging state out of a trait value, total: a value that is
    /// not an enum, or whose ordinal is outside this set, answers [#UNKNOWN].
    ///
    /// #### Parameters
    ///
    /// - `value`: a value read from [Trait#BATTERY_CHARGING], or `null`
    ///
    /// #### Returns
    ///
    /// the state, never `null`
    public static ChargingState of(TraitValue value) {
        if (value == null || value.getKind() != TraitValueKind.ENUM) {
            return UNKNOWN;
        }
        int ordinal = value.getEnumOrdinal();
        ChargingState[] all = values();
        if (ordinal < 0 || ordinal >= all.length) {
            return UNKNOWN;
        }
        return all[ordinal];
    }
}
