package com.codename1.ui.graph;

import com.codename1.graph.GraphEdge;
import com.codename1.graph.GraphGroup;
import com.codename1.graph.GraphModel;
import com.codename1.graph.GraphNode;
import com.codename1.graph.GraphPort;
import com.codename1.util.UndoableEdit;

import java.util.ArrayList;
import java.util.List;

/// The steps a `NodeGraph` records in an `UndoHistory`. Each replays through
/// the model, the way the user's own edit went, so a listening view repaints.
final class GraphEdits { // NOPMD MissingStaticMethodInNonInstantiatableClass - a holder of the edit classes
    private GraphEdits() {
    }

    /// A node went from one place to another.
    static final class Move extends UndoableEdit {
        private final GraphModel model;
        private final GraphNode node;
        private final double fromX;
        private final double fromY;
        private final double toX;
        private final double toY;

        Move(GraphModel model, GraphNode node, double fromX, double fromY, double toX, double toY) {
            this.model = model;
            this.node = node;
            this.fromX = fromX;
            this.fromY = fromY;
            this.toX = toX;
            this.toY = toY;
        }

        @Override
        public void undo() {
            model.moveNode(node, fromX, fromY);
        }

        @Override
        public void redo() {
            model.moveNode(node, toX, toY);
        }

        @Override
        public String getName() {
            return "Move";
        }
    }

    /// Edges lost the routes an arrangement gave them when a node moved.
    static final class Routes extends UndoableEdit {
        private final NodeGraph graph;
        private final List<GraphEdge> edges;
        private final List<double[]> before;
        private final List<double[]> after;

        Routes(NodeGraph graph, List<GraphEdge> edges, List<double[]> before, List<double[]> after) {
            this.graph = graph;
            this.edges = edges;
            this.before = before;
            this.after = after;
        }

        private void apply(List<double[]> routes) {
            int n = edges.size();
            for (int i = 0; i < n; i++) {
                edges.get(i).setRoute(routes == null ? null : routes.get(i));
            }
            // a route is not an edit of the model, so no listener hears of it
            graph.repaint();
        }

        @Override
        public void undo() {
            apply(before);
        }

        @Override
        public void redo() {
            apply(after);
        }

        @Override
        public String getName() {
            return "Move";
        }
    }

    /// A node took a new size, and with a leading or top corner a new place.
    static final class Resize extends UndoableEdit {
        private final GraphModel model;
        private final GraphNode node;
        private final double[] from;
        private final double[] to;

        Resize(GraphModel model, GraphNode node, double[] from, double[] to) {
            this.model = model;
            this.node = node;
            this.from = from;
            this.to = to;
        }

        private void apply(double[] b) {
            model.moveNode(node, b[0], b[1]);
            model.resizeNode(node, b[2], b[3]);
        }

        @Override
        public void undo() {
            apply(from);
        }

        @Override
        public void redo() {
            apply(to);
        }

        @Override
        public String getName() {
            return "Resize";
        }
    }

    /// Nodes and edges were added, or with `removal` taken out.
    static final class Presence extends UndoableEdit {
        private final GraphModel model;
        private final List<GraphNode> nodes;
        private final List<GraphEdge> edges;
        private final boolean removal;
        private final String name;
        // where each was in the model before a removal, or null: at the end
        private final int[] nodeAt;
        private final int[] edgeAt;

        Presence(GraphModel model, List<GraphNode> nodes, List<GraphEdge> edges, boolean removal, String name) {
            this(model, nodes, edges, removal, name, null, null);
        }

        /// A removal, with the model's nodes and edges as they were before
        /// it: undo puts each back where it was, so what was drawn under
        /// another node is under it again.
        Presence(GraphModel model, List<GraphNode> nodes, List<GraphEdge> edges, boolean removal, String name,
                 List<GraphNode> nodesBefore, List<GraphEdge> edgesBefore) {
            this.model = model;
            this.removal = removal;
            this.name = name;
            if (nodesBefore == null) {
                this.nodes = nodes;
                this.edges = edges;
                nodeAt = null;
                edgeAt = null;
                return;
            }
            // in the order they had, so that each index is right when it is used
            this.nodes = new ArrayList<GraphNode>();
            this.edges = new ArrayList<GraphEdge>();
            nodeAt = new int[nodes.size()];
            edgeAt = new int[edges.size()];
            int k = 0;
            for (int i = 0; i < nodesBefore.size() && k < nodeAt.length; i++) {
                if (nodes.contains(nodesBefore.get(i))) {
                    this.nodes.add(nodesBefore.get(i));
                    nodeAt[k++] = i;
                }
            }
            k = 0;
            for (int i = 0; i < edgesBefore.size() && k < edgeAt.length; i++) {
                if (edges.contains(edgesBefore.get(i))) {
                    this.edges.add(edgesBefore.get(i));
                    edgeAt[k++] = i;
                }
            }
        }

        private void put() {
            int n = nodes.size();
            for (int i = 0; i < n; i++) {
                model.insertNode(nodes.get(i), nodeAt == null ? -1 : nodeAt[i]);
            }
            n = edges.size();
            for (int i = 0; i < n; i++) {
                model.insertEdge(edges.get(i), edgeAt == null ? -1 : edgeAt[i]);
            }
        }

        private void take() {
            for (int i = edges.size() - 1; i >= 0; i--) {
                model.removeEdge(edges.get(i));
            }
            for (int i = nodes.size() - 1; i >= 0; i--) {
                model.removeNode(nodes.get(i));
            }
        }

        @Override
        public void undo() {
            if (removal) {
                put();
            } else {
                take();
            }
        }

        @Override
        public void redo() {
            if (removal) {
                take();
            } else {
                put();
            }
        }

        @Override
        public String getName() {
            return name;
        }
    }

    /// A group was collapsed or expanded.
    static final class Collapse extends UndoableEdit {
        private final GraphModel model;
        private final GraphGroup group;
        private final boolean to;

        Collapse(GraphModel model, GraphGroup group, boolean to) {
            this.model = model;
            this.group = group;
            this.to = to;
        }

        @Override
        public void undo() {
            model.setCollapsed(group, !to);
        }

        @Override
        public void redo() {
            model.setCollapsed(group, to);
        }

        @Override
        public String getName() {
            return to ? "Collapse" : "Expand";
        }
    }

    /// An arrangement placed every node. Like the arrangement itself, this
    /// is not offered to the model as a user's move: nodes a model refuses to
    /// move are placed directly.
    static final class Arrange extends UndoableEdit {
        private final GraphModel model;
        private final List<GraphNode> nodes;
        private final double[] from;
        private final double[] to;
        private final List<GraphEdge> edges;
        private final List<double[]> routesFrom;
        private final List<double[]> routesTo;

        Arrange(GraphModel model, List<GraphNode> nodes, double[] from, double[] to,
                List<GraphEdge> edges, List<double[]> routesFrom, List<double[]> routesTo) {
            this.model = model;
            this.nodes = nodes;
            this.from = from;
            this.to = to;
            this.edges = edges;
            this.routesFrom = routesFrom;
            this.routesTo = routesTo;
        }

        private void apply(double[] at, List<double[]> routes) {
            int n = edges.size();
            for (int i = 0; i < n; i++) {
                edges.get(i).setRoute(routes.get(i));
            }
            n = nodes.size();
            for (int i = 0; i < n; i++) {
                GraphNode node = nodes.get(i);
                if (!model.moveNode(node, at[i * 2], at[i * 2 + 1])) {
                    node.setPosition(at[i * 2], at[i * 2 + 1]);
                }
            }
        }

        @Override
        public void undo() {
            apply(from, routesFrom);
        }

        @Override
        public void redo() {
            apply(to, routesTo);
        }

        @Override
        public String getName() {
            return "Arrange";
        }
    }
