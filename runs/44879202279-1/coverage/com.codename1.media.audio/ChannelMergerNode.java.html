package com.codename1.media.audio;

/// Merges mono inputs into one output with a channel per input, the Web Audio
/// `ChannelMergerNode`. Each input is mixed down to mono first.
public final class ChannelMergerNode extends AudioNode {
    ChannelMergerNode(BaseAudioContext context, int inputs) {
        super(context, ChannelSplitterNode.checkCount(inputs), 1, 1, ChannelCountMode.EXPLICIT,
                ChannelInterpretation.SPEAKERS);
    }

    @Override
    void checkChannelCount(int count) {
        if (count != 1) {
            throw new IllegalArgumentException("A merger's channel count is 1");
        }
    }

    @Override
    void checkChannelCountMode(ChannelCountMode mode) {
        if (mode != ChannelCountMode.EXPLICIT) {
            throw new IllegalArgumentException("A merger's channel count mode is EXPLICIT");
        }
    }

    @Override
    void process(long q) {
        AudioBus out = outBuses[0];
        int n = inBuses.length;
        out.setChannels(n);
        for (int k = 0; k < n; k++) {
            System.arraycopy(inBuses[k].data[0], 0, out.data[k], 0, BaseAudioContext.RENDER_QUANTUM);
        }
    }
}
