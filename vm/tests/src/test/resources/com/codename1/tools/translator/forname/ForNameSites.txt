// Stand-ins for the Class.forName call sites ReachabilityCull.FOR_NAME_SITES narrows:
// same class names, method names and result types as the real core and iOS port, so
// the translator's table matches them, but small enough to translate in seconds and
// with nothing else in the program calling forName. One Site* main per site makes
// exactly that site live. Each Seed* class is created ONLY by name -- no NEW, no class
// literal, no string constant spelling it -- and is kept in the program by a static call.

import com.codename1.background.BackgroundWorker;
import com.codename1.location.GeofenceListener;
import com.codename1.location.LocationListener;
import com.codename1.system.NativeInterface;
import com.codename1.testing.UnitTest;

interface CullProbe {
    void probe();
}

interface SeedNative extends NativeInterface {
}

class SeedGeofence implements GeofenceListener {
    static void keep() {
    }

    public void onEntered(String id) {
        System.out.println("CASE|geofence|" + id);
    }
}

class SeedLocation implements LocationListener {
    static void keep() {
    }

    public void providerStateChanged(int newState) {
        System.out.println("CASE|location|" + newState);
    }
}

class SeedWorker implements BackgroundWorker {
    static void keep() {
    }

    public void performWork(String workId) {
        System.out.println("CASE|worker|" + workId);
    }
}

class SeedTest implements UnitTest {
    static void keep() {
    }

    public boolean shouldExecuteOnEDT() {
        return true;
    }
}

class SeedNativeImpl implements SeedNative {
    static void keep() {
    }

    public boolean isSupported() {
        return true;
    }
}

/// Assignable to none of the narrowed types: only an unnarrowed site may keep it.
class SeedOther implements CullProbe {
    static void keep() {
    }

    public void probe() {
        System.out.println("CASE|other|probed");
    }
}

public class Seeds {
    /// Keeps every seed class in the program without allocating any of them.
    public static void keepAll() {
        SeedGeofence.keep();
        SeedLocation.keep();
        SeedWorker.keep();
        SeedTest.keep();
        SeedNativeImpl.keep();
        SeedOther.keep();
    }

    /// Calls every seed interface method, so each is reached and the cull has to decide
    /// between a body and a stub for every implementation.
    public static void poke(Object o) {
        if (o instanceof GeofenceListener) {
            ((GeofenceListener) o).onEntered("x");
        }
        if (o instanceof LocationListener) {
            ((LocationListener) o).providerStateChanged(1);
        }
        if (o instanceof BackgroundWorker) {
            ((BackgroundWorker) o).performWork("x");
        }
        if (o instanceof UnitTest) {
            System.out.println("CASE|test|" + ((UnitTest) o).shouldExecuteOnEDT());
        }
        if (o instanceof NativeInterface) {
            System.out.println("CASE|native|" + ((NativeInterface) o).isSupported());
        }
        if (o instanceof CullProbe) {
            ((CullProbe) o).probe();
        }
    }

    /// A name no constant spells, so only a forName seed can account for the class.
    public static String name(String reversed) {
        return new StringBuilder(reversed).reverse().toString();
    }
}

/// no-argument constructor has to count as allocated.

//// FILE KotlinLikeReflection.java
/// Stands in for kotlin.jvm.internal.Reflection's own `static {}`, which loads
/// "kotlin.reflect.jvm.internal.ReflectionFactoryImpl" -- a class kotlin-reflect defines
/// and that is therefore never part of the translated universe, since kotlin-reflect is
/// never bundled. The constant reaches Class.forName directly, with no intervening local
/// or field (the exact shape javap shows for the real class): ldc "..."; invokestatic
/// Class.forName. See ReachabilityCull.Allocation.literalForNameTargets.
public class KotlinLikeReflection {
    private static final Object factory;
    static {
        Object f;
        try {
            f = Class.forName("not.bundled.reflect.ReflectionFactoryImpl").newInstance();
        } catch (Exception e) {
            f = null;
        }
        factory = f;
    }

    public static Object getFactory() {
        return factory;
    }
}
//// FILE SiteKotlinLikeReflection.java
/// A forName fed only a literal naming an absent class must narrow to NOTHING: none of
/// the Seeds may be kept by this alone, unlike SiteUnlisted above.
public class SiteKotlinLikeReflection {
    public static void main(String[] args) throws Exception {
        Seeds.keepAll();
        System.out.println("CASE|factory|" + KotlinLikeReflection.getFactory());
    }
}
