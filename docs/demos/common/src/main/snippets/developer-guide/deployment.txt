// tag::deploy-native-tls[]
# Once: apt-get install -y certbot && certbot certonly --standalone -d app.example.com
# /etc/myapp/env, when the server terminates TLS itself. Remove CN1_SERVER_ADDRESS,
# CN1_SERVER_FORWARDHEADERS, CN1_SERVER_TRUSTEDPROXIES and CN1_SESSION_SECURE:
# there is no proxy, and a server bound to 127.0.0.1 can't be reached on 443.
CN1_SERVER_PORT=443
CN1_SERVER_TLS_CERTIFICATE=/etc/myapp/tls/fullchain.pem
CN1_SERVER_TLS_KEY=/etc/myapp/tls/privkey.pem

# myapp.service, under [Service]: a port below 1024 without running as root
AmbientCapabilities=CAP_NET_BIND_SERVICE

# /etc/letsencrypt/renewal-hooks/deploy/myapp -- chmod 755. The server reads the
# files when it starts, so a renewed certificate needs a restart.
#!/bin/sh
install -o myapp -g myapp -m 600 -D \
    /etc/letsencrypt/live/app.example.com/fullchain.pem /etc/myapp/tls/fullchain.pem
install -o myapp -g myapp -m 600 -D \
    /etc/letsencrypt/live/app.example.com/privkey.pem /etc/myapp/tls/privkey.pem
systemctl restart myapp
// end::deploy-native-tls[]

// tag::deploy-dockerfile[]
# Dockerfile -- the binary was built on ubuntu:24.04, so it runs on ubuntu:24.04
FROM ubuntu:24.04
RUN apt-get update \
 && apt-get install -y --no-install-recommends libssl3t64 libnghttp2-14 libcurl4t64 ca-certificates \
 && rm -rf /var/lib/apt/lists/* \
 && useradd --system --home /app myapp
WORKDIR /app
COPY backend/target/myapp-backend ./myapp
COPY backend/application.properties ./
COPY backend/target/webapp ./webapp
USER myapp
EXPOSE 8080
ENTRYPOINT ["/app/myapp"]
// end::deploy-dockerfile[]

// tag::deploy-github-actions[]
# .github/workflows/deploy.yml
name: deploy
on:
  push:
    branches: [main]
jobs:
  server:
    runs-on: ubuntu-24.04          # the server's release, so the libraries match
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with: { distribution: temurin, java-version: '17', cache: maven }
      - run: sudo apt-get update && sudo apt-get install -y clang libssl-dev libnghttp2-dev libcurl4-openssl-dev
      - run: ./mvnw -B -pl backend -Dcodename1.platform=backend cn1:backend-webapp
      - run: ./mvnw -B -pl backend -Dcodename1.platform=backend cn1:backend-package
      - run: |
          mkdir -p ~/.ssh && echo "$DEPLOY_KEY" > ~/.ssh/id_ed25519 && chmod 600 ~/.ssh/id_ed25519
          echo "$KNOWN_HOSTS" > ~/.ssh/known_hosts
          sh deploy.sh
        env:
          DEPLOY_KEY: ${{ secrets.DEPLOY_KEY }}
          KNOWN_HOSTS: ${{ secrets.KNOWN_HOSTS }}   # ssh-keyscan app.example.com, checked once by hand
// end::deploy-github-actions[]
