    &state=<random>

// tag::extending-bazaar-function-discount[]
for line in lines {
  if line.quantity >= 2 { emit line_discount(line.index, line.total / 4, "Pair deal"); }
}
if cart.quantity >= 3 { emit discount(100, "Three and up"); }
// end::extending-bazaar-function-discount[]

// tag::extending-bazaar-function-shipping[]
for r in rates {
  if r.name == "Express" { emit rename(r.id, "Rocket"); emit reprice(r.id, 1); emit move(r.id, 0); }
  else { emit hide(r.id); }
}
// end::extending-bazaar-function-shipping[]

// tag::extending-bazaar-function-payment[]
for m in methods {
  if m.type == "card" && cart.subtotal >= 2000 { emit hide(m.type); }
  if m.type == "affirm" { emit move(m.type, 0); }
}
// end::extending-bazaar-function-payment[]
